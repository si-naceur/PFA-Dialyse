import json
import re
import torch
from fastapi import FastAPI, UploadFile, File
from transformers import Qwen2_5_VLForConditionalGeneration, AutoProcessor
from PIL import Image

MODEL = "Qwen/Qwen2.5-VL-3B-Instruct"

app = FastAPI(title="PFA Dialyse - Qwen Vision API")

print("Loading Qwen on GPU...")

model = Qwen2_5_VLForConditionalGeneration.from_pretrained(
    MODEL,
    torch_dtype=torch.float16,
    device_map="auto",
    low_cpu_mem_usage=True,
)

processor = AutoProcessor.from_pretrained(
    MODEL,
    min_pixels=256 * 256,
    max_pixels=1024 * 1024,
)

print("Qwen loaded successfully.")


PROMPT = """
Analyze the entire dialysis machine screen carefully.

IMPORTANT:
The blood flow value Qb is displayed as a numeric value on the screen.
Do not return null for Qb unless you have genuinely inspected the whole image.

Extract these values:
- qb: blood flow
- pv: venous pressure
- pa: arterial pressure
- ptm: transmembrane pressure
- uf_volume: ultrafiltration volume
- uf_rate: ultrafiltration rate
- heparin: heparin dose or rate

Return ONLY valid JSON:

{
  "qb": number or null,
  "pv": number or null,
  "pa": number or null,
  "ptm": number or null,
  "uf_volume": number or null,
  "uf_rate": number or null,
  "heparin": number or null
}

Rules:
- Analyze the entire image.
- Read values directly from the screen.
- Do not guess.
- If a value is not clearly visible, return null.
- Preserve negative signs exactly.
- Do not calculate values.
- Do not confuse Qb with another numerical value.
- Return numbers only, without units.
- Return JSON only.
"""


@app.get("/health")
def health():
    return {
        "status": "ok",
        "model": MODEL,
        "gpu": torch.cuda.get_device_name(0) if torch.cuda.is_available() else "CPU"
    }


@app.post("/analyze")
async def analyze(file: UploadFile = File(...)):

    image_bytes = await file.read()

    image_path = "received_frame.jpg"

    with open(image_path, "wb") as f:
        f.write(image_bytes)

    image = Image.open(image_path).convert("RGB")

    conversation = [
        {
            "role": "user",
            "content": [
                {
                    "type": "image",
                    "image": image,
                },
                {
                    "type": "text",
                    "text": PROMPT,
                },
            ],
        }
    ]

    inputs = processor.apply_chat_template(
        conversation,
        add_generation_prompt=True,
        tokenize=True,
        return_dict=True,
        return_tensors="pt",
    )

    inputs = inputs.to(model.device)

    print("Running Qwen inference...")

    with torch.inference_mode():
        generated_ids = model.generate(
            **inputs,
            max_new_tokens=150,
            do_sample=False,
        )

    generated_ids = [
        output_ids[len(input_ids):]
        for input_ids, output_ids in zip(
            inputs.input_ids,
            generated_ids
        )
    ]

    output = processor.batch_decode(
        generated_ids,
        skip_special_tokens=True,
        clean_up_tokenization_spaces=False,
    )[0]

    print("\n========== QWEN RAW ==========")
    print(output)

    cleaned = output.strip()

    if cleaned.startswith("```"):
        cleaned = re.sub(r"```json|```", "", cleaned).strip()

    try:
        data = json.loads(cleaned)
    except json.JSONDecodeError:
        return {
            "success": False,
            "error": "Qwen did not return valid JSON",
            "raw": output,
        }

    print("\n========== RESULT ==========")
    print(json.dumps(data, indent=2))

    return {
        "success": True,
        "data": data
    }
